	private function EitherEndian2Int($byteword, $signed=false) {
		if ($this->getid3->info['fileformat'] == 'riff') {
			return getid3_lib::LittleEndian2Int($byteword, $signed);
		}
		return getid3_lib::BigEndian2Int($byteword, false, $signed);
	}

	/**
	 * Division that mimics the pre-PHP-8 behaviour of returning INF / -INF / NAN
	 * on division by zero instead of throwing a DivisionByZeroError.
	 */
	private static function legacyDivide($dividend, $divisor) {
		if ($divisor == 0) {
			return fdiv((float) $dividend, (float) $divisor);
		}
		return $dividend / $divisor;
	}

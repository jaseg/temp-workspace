	private function EitherEndian2Int($byteword, $signed=false) {
		if (($this->getid3->info['fileformat'] ?? null) == 'riff') {
			return getid3_lib::LittleEndian2Int($byteword, $signed);
		}
		return getid3_lib::BigEndian2Int($byteword, false, $signed);
	}

	/**
	 * Parses an iXML "numerator/denominator" ratio string the same way the
	 * legacy `@list($n, $d) = explode('/', $v); $n / ($d ? $d : 1000)` code did,
	 * without raising TypeError / DivisionByZeroError on PHP 8.
	 *
	 * @param mixed $ratio
	 * @return int|float
	 */
	private static function parseIXMLratio(mixed $ratio): int|float {
		$parts       = (is_scalar($ratio) ? explode('/', (string) $ratio) : []);
		$numerator   = $parts[0] ?? null;
		$denominator = $parts[1] ?? null;
		return self::legacyDivide($numerator, ($denominator ?: 1000));
	}

	/**
	 * Converts a value to a number following pre-PHP-8 arithmetic coercion rules
	 * (non-numeric strings become 0, leading-numeric strings use their numeric prefix).
	 *
	 * @param mixed $value
	 * @return int|float
	 */
	private static function legacyNumber(mixed $value): int|float {
		if (is_int($value) || is_float($value)) {
			return $value;
		}
		if ($value === null || is_bool($value)) {
			return (int) $value;
		}
		if (!is_scalar($value)) {
			return 0;
		}
		$value = (string) $value;
		if (is_numeric($value)) {
			return $value + 0;
		}
		if (preg_match('#^[ \\t\\n\\r\\v\\f]*[+-]?(?:[0-9]+(?:\\.[0-9]*)?|\\.[0-9]+)(?:[eE][+-]?[0-9]+)?#', $value, $matches)) {
			return trim($matches[0]) + 0;
		}
		return 0;
	}

	/**
	 * Division with pre-PHP-8 semantics: division by zero yields INF / -INF / NAN
	 * instead of throwing DivisionByZeroError.
	 *
	 * @param mixed $dividend
	 * @param mixed $divisor
	 * @return int|float
	 */
	private static function legacyDivide(mixed $dividend, mixed $divisor): int|float {
		$dividend = self::legacyNumber($dividend);
		$divisor  = self::legacyNumber($divisor);
		if ($divisor == 0) {
			return fdiv($dividend, $divisor);
		}
		return $dividend / $divisor;
	}
